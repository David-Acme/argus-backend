cmake_minimum_required(VERSION 3.25)
include(${CMAKE_CURRENT_SOURCE_DIR}/../../../cmake/argus-module.cmake)
set(CMAKE_FIND_PACKAGE_TARGETS_GLOBAL TRUE)
argus_contracts_substrate()
if(NOT ARGUS_SYSTEM_PROTOBUF AND NOT TARGET gRPC::grpc++)
    find_package(gRPC CONFIG REQUIRED)
endif()
if(NOT TARGET contract::tts)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../contracts/tts
                     ${CMAKE_CURRENT_BINARY_DIR}/contracts)
endif()
argus_tts_rpc_contract()
argus_module(NAME tts-rpc-client SOURCES tts-client.cc INCLUDES .)
target_link_libraries(argus_tts-rpc-client PUBLIC contract::tts PRIVATE argus::sdk-tts-wire)
target_compile_options(argus_tts-rpc-client PRIVATE -Wall -Wextra)
